TreeCare

An iPhone app and DIY Arduino sensor that help municipalities find and map dead or dying trees before they become a falling hazard.

SwiftUI + SwiftData ESP32 + Bluetooth LE On-device Vision Prototype

What it does

TreeCare gives a municipal field crew two independent ways to flag a dead tree, both of which log the tree's GPS coordinates to a shared, on-device map. The goal is to make tree-risk surveys faster and less manual than walking a neighborhood with a clipboard.

Two ways to detect a dead tree

Sensor mode

An ESP32 microcontroller drives a pair of handheld piezo probes pressed against opposite sides of a trunk. It times how long a sonic pulse takes to cross from one side to the other — a simplified version of professional sonic tomography. Healthy wood transmits sound faster than decayed or hollow wood. The ESP32 talks to the app over Bluetooth Low Energy and the reading is logged instantly with GPS coordinates.

Aerial mode

Point the phone's camera — handheld, or mounted on a drone the municipality already owns — at tree canopy and leave it running. The app continuously analyzes the live feed on-device: a scene classifier confirms it's actually looking at vegetation, subject segmentation isolates the tree from its background, and a color/saturation/texture model scores the canopy for signs of dieback. A sustained "dead-looking" detection logs the tree automatically — no photo-taking required.

How the pieces connect

  • ESP32 pulses a piezo transducer through the trunk and times the response on a second piezo
  • Result is classified dead / alive / uncertain on the microcontroller itself
  • ESP32 broadcasts the reading over a custom Bluetooth LE GATT service
  • iPhone app connects as a BLE central, requests readings, and tags them with CoreLocation coordinates
  • Aerial scans run entirely on-device using Apple's Vision framework — no cloud, no external API
  • Every logged tree lands in one shared map and list view, regardless of which method found it

Built with

Swift & SwiftUI SwiftData CoreBluetooth CoreLocation AVFoundation Vision framework ESP32 / Arduino C++ Bluetooth LE (GATT) XcodeGen

import Foundation
import SwiftData
import CoreLocation

/// How a tree entry was logged.
enum TreeSource: String, Codable {
    case sensor   // ESP32 trunk sonic sensor via Bluetooth
    case aerial   // phone/drone camera + on-device canopy analysis
    case manual   // manually entered, no automated reading
}

/// The health call for a logged tree. Both the sensor and aerial modes log
/// `.dead` directly - there's no separate "unconfirmed" tier.
enum TreeHealthStatus: String, Codable {
    case dead
    case alive
    case uncertain

    /// Maps the ESP32 sensor's 0/1/2 status byte to a case.
    init(sensorStatus: Int) {
        switch sensorStatus {
        case 1: self = .dead
        case 0: self = .alive
        default: self = .uncertain
        }
    }

    /// Human-readable label (kept as a small convenience over
    /// `rawValue.capitalized`).
    var displayName: String {
        switch self {
        case .dead: return "Dead"
        case .alive: return "Alive"
        case .uncertain: return "Uncertain"
        }
    }
}

/// One logged tree observation. This is the local (on-device) database -
/// SwiftData persists these to a SQLite store in the app's sandbox.
@Model
final class TreeRecord {
    var id: UUID
    var timestamp: Date
    var latitude: Double
    var longitude: Double
    var source: TreeSource
    var status: TreeHealthStatus

    /// Sonic pulse velocity in m/s, only set for .sensor readings.
    var sonicVelocity: Double?
    /// Classifier confidence 0-1, only set for .aerial readings.
    var confidence: Double?
    var notes: String?
    /// Filename of a saved JPEG in the app's Documents directory, if any.
    var photoFileName: String?

    init(
        id: UUID = UUID(),
        timestamp: Date,
        latitude: Double,
        longitude: Double,
        source: TreeSource,
        status: TreeHealthStatus,
        sonicVelocity: Double? = nil,
        confidence: Double? = nil,
        notes: String? = nil,
        photoFileName: String? = nil
    ) {
        self.id = id
        self.timestamp = timestamp
        self.latitude = latitude
        self.longitude = longitude
        self.source = source
        self.status = status
        self.sonicVelocity = sonicVelocity
        self.confidence = confidence
        self.notes = notes
        self.photoFileName = photoFileName
    }

    var coordinate: CLLocationCoordinate2D {
        CLLocationCoordinate2D(latitude: latitude, longitude: longitude)
    }
}
BLEManager.swiftBluetooth LE central 
import Foundation
import CoreBluetooth

/// One reading from the ESP32 trunk sensor, decoded from its BLE JSON payload.
/// status: 0 = alive, 1 = dead, 2 = uncertain (see DeadTreeSensor.ino).
struct SensorReading: Codable, Equatable {
    let v: Double
    let amp: Int
    let status: Int
}

/// Talks to the ESP32 "DeadTreeSensor" over Bluetooth LE using CoreBluetooth.
///
/// NOTE: CoreBluetooth does not work in the iOS Simulator - you need a real
/// iPhone to test against real hardware. Use `simulateReading(status:)` to
/// exercise the rest of the app (UI, logging, map) without hardware.
final class BLEManager: NSObject, ObservableObject {
    static let serviceUUID = CBUUID(string: "A1B2C3D4-0001-1000-8000-00805F9B34FB")
    static let readingCharUUID = CBUUID(string: "A1B2C3D4-0002-1000-8000-00805F9B34FB")
    static let commandCharUUID = CBUUID(string: "A1B2C3D4-0003-1000-8000-00805F9B34FB")

    @Published var isBluetoothReady = false
    @Published var discoveredPeripherals: [CBPeripheral] = []
    @Published var connectedPeripheral: CBPeripheral?
    @Published var latestReading: SensorReading?
    @Published var connectionState: String = "Disconnected"

    private var centralManager: CBCentralManager!
    private var readingCharacteristic: CBCharacteristic?
    private var commandCharacteristic: CBCharacteristic?

    override init() {
        super.init()
        centralManager = CBCentralManager(delegate: self, queue: nil)
    }

    func startScanning() {
        guard centralManager.state == .poweredOn else { return }
        discoveredPeripherals.removeAll()
        connectionState = "Scanning..."
        centralManager.scanForPeripherals(withServices: [Self.serviceUUID], options: nil)
    }

    func stopScanning() {
        centralManager.stopScan()
    }

    func connect(_ peripheral: CBPeripheral) {
        connectionState = "Connecting..."
        centralManager.connect(peripheral, options: nil)
    }

    func disconnect() {
        if let peripheral = connectedPeripheral {
            centralManager.cancelPeripheralConnection(peripheral)
        }
    }

    /// Asks the connected sensor to take a fresh sonic pulse reading.
    func requestPulseReading() {
        guard let peripheral = connectedPeripheral, let characteristic = commandCharacteristic else { return }
        let command = "PULSE".data(using: .utf8)!
        peripheral.writeValue(command, for: characteristic, type: .withResponse)
    }

    /// Fakes a reading with no hardware attached, for UI development/testing
    /// in the Simulator or before you've built the Arduino sensor.
    /// status: 0 = alive, 1 = dead, 2 = uncertain
    func simulateReading(status: Int) {
        let velocity: Double = status == 1 ? 650 : (status == 0 ? 1450 : 950)
        latestReading = SensorReading(v: velocity, amp: Int.random(in: 400...900), status: status)
        connectionState = "Simulated (no hardware connected)"
    }
}

extension BLEManager: CBCentralManagerDelegate {
    func centralManagerDidUpdateState(_ central: CBCentralManager) {
        isBluetoothReady = central.state == .poweredOn
        if central.state != .poweredOn {
            connectionState = "Bluetooth unavailable"
        }
    }

    func centralManager(_ central: CBCentralManager, didDiscover peripheral: CBPeripheral, advertisementData: [String: Any], rssi RSSI: NSNumber) {
        if !discoveredPeripherals.contains(where: { $0.identifier == peripheral.identifier }) {
            discoveredPeripherals.append(peripheral)
        }
    }

    func centralManager(_ central: CBCentralManager, didConnect peripheral: CBPeripheral) {
        connectedPeripheral = peripheral
        connectionState = "Connected: \(peripheral.name ?? "sensor")"
        peripheral.delegate = self
        peripheral.discoverServices([Self.serviceUUID])
        stopScanning()
    }

    func centralManager(_ central: CBCentralManager, didFailToConnect peripheral: CBPeripheral, error: Error?) {
        connectionState = "Failed to connect"
    }

    func centralManager(_ central: CBCentralManager, didDisconnectPeripheral peripheral: CBPeripheral, error: Error?) {
        connectedPeripheral = nil
        readingCharacteristic = nil
        commandCharacteristic = nil
        connectionState = "Disconnected"
    }
}

extension BLEManager: CBPeripheralDelegate {
    func peripheral(_ peripheral: CBPeripheral, didDiscoverServices error: Error?) {
        guard let services = peripheral.services else { return }
        for service in services where service.uuid == Self.serviceUUID {
            peripheral.discoverCharacteristics([Self.readingCharUUID, Self.commandCharUUID], for: service)
        }
    }

    func peripheral(_ peripheral: CBPeripheral, didDiscoverCharacteristicsFor service: CBService, error: Error?) {
        guard let characteristics = service.characteristics else { return }
        for characteristic in characteristics {
            if characteristic.uuid == Self.readingCharUUID {
                readingCharacteristic = characteristic
                peripheral.setNotifyValue(true, for: characteristic)
            } else if characteristic.uuid == Self.commandCharUUID {
                commandCharacteristic = characteristic
            }
        }
    }

    func peripheral(_ peripheral: CBPeripheral, didUpdateValueFor characteristic: CBCharacteristic, error: Error?) {
        guard characteristic.uuid == Self.readingCharUUID, let data = characteristic.value else { return }
        if let reading = try? JSONDecoder().decode(SensorReading.self, from: data) {
            DispatchQueue.main.async {
                self.latestReading = reading
            }
        }
    }
}

TreeHealthVisionAnalyzer.swiftCanopy classifier 
import UIKit
import CoreVideo

/// Result of classifying a canopy view (single photo or one live camera frame).
struct AerialClassification {
    let status: TreeHealthStatus
    /// How confident the classifier is in `status`, 0-1.
    let confidence: Double
    /// The combined "how dead does this look" score (0-1) that `status` was
    /// derived from - a blend of `colorDeadFraction` and `porosityFraction`.
    let deadCanopyFraction: Double
    /// Sub-score: fraction of scored pixels that are both non-green AND
    /// desaturated (i.e. brown/gray, not vibrant autumn color). Exposed
    /// separately so the live UI can show it while you calibrate.
    let colorDeadFraction: Double
    /// Sub-score: fraction of scored pixels that are bright enough to be sky
    /// showing through gaps in a sparse/bare canopy. Exposed for calibration
    /// too.
    let porosityFraction: Double
}

/// Anything that can look at a top-down canopy image/frame and call
/// dead/alive/uncertain. This indirection exists so the v1 heuristic below
/// can be swapped for a real trained Core ML model later without touching
/// the views.
protocol TreeHealthClassifying {
    /// Classify a static image (kept for reuse/testing; the live Aerial tab
    /// no longer uses this path directly - see classify(pixelBuffer:foregroundMask:)).
    func classify(image: UIImage) -> AerialClassification
    /// Classify one live camera frame, called continuously while scanning.
    /// Expects 32BGRA pixel format (that's what CameraLiveScanner requests).
    /// `foregroundMask`, if provided, is an 8-bit single-channel mask the
    /// same size as `pixelBuffer` (0 = background, 255 = foreground subject,
    /// as produced by CanopySegmenter) - when present, only pixels inside
    /// the masked subject are scored, so background sky/ground/pavement
    /// don't get counted as canopy.
    func classify(pixelBuffer: CVPixelBuffer, foregroundMask: CVPixelBuffer?) -> AerialClassification
}

extension TreeHealthClassifying {
    /// Convenience overload for callers with no mask available.
    func classify(pixelBuffer: CVPixelBuffer) -> AerialClassification {
        classify(pixelBuffer: pixelBuffer, foregroundMask: nil)
    }
}

/// v1 heuristic classifier: no machine learning model, no training data
/// required. It looks at three color/brightness-derived signals per pixel,
/// none of which need a trained model:
///
/// 1. **Greenness** (Excess Green Index) - live, chlorophyll-rich foliage is
///    disproportionately green from above.
/// 2. **Saturation** - dead/bare wood is desaturated gray-brown. This is what
///    keeps vibrant (highly saturated) autumn reds/oranges/yellows from
///    being misread as "dead" just because they're not green.
/// 3. **Porosity** (a texture proxy) - dense healthy foliage occludes the
///    sky almost completely when viewed from directly above; a sparse or
///    bare canopy lets bright sky show through the gaps between branches.
///    This is measured as the fraction of scored pixels that are bright
///    enough to plausibly be sky glimpsed through the canopy.
///
/// This is a reasonable, honest step up from color alone, but it is still
/// NOT as accurate as a real trained model. Research on this exact problem
/// (standing dead tree detection in UAV RGB imagery) uses trained deep
/// models - e.g. Faster R-CNN, YOLOv8/YOLO11 instance segmentation, or
/// U-Net-based segmentation - trained on labeled aerial photos, and reaches
/// much higher accuracy than any hand-written heuristic can. See README.md
/// for how to train one and drop it in via a new `TreeHealthClassifying`
/// conformance - nothing else in the app depends on this specific class.
final class HeuristicCanopyClassifier: TreeHealthClassifying {
    /// Sample every Nth pixel in each dimension - full-resolution frames
    /// don't need per-pixel analysis for this heuristic, and this keeps
    /// classification fast enough to run continuously on-device.
    private let sampleStride = 4

    /// A pixel is counted as scoreable (not sky/pavement/deep shadow) once
    /// its channel sum passes this brightness floor.
    private let canopyBrightnessFloor: Double = 60

    /// Excess Green below this value is treated as "not distinctly green".
    private let deadPixelExGThreshold: Double = 10

    /// Saturation (0-1) below this is treated as "desaturated" - a pixel
    /// only counts toward colorDeadFraction if it's BOTH non-green and
    /// desaturated, which is what rules out vibrant fall foliage.
    private let deadPixelMaxSaturation: Double = 0.35

    /// Channel sum above this is treated as "bright enough to be sky
    /// glimpsed through canopy gaps" for the porosity signal.
    private let skyBrightnessThreshold: Double = 650

    /// How much weight the color signal vs. the porosity signal gets in the
    /// combined dead score. Color is weighted higher since it's the more
    /// direct signal; porosity is noisier (glare, gaps in framing, etc.).
    private let colorWeight = 0.65
    private let porosityWeight = 0.35

    /// Foreground mask pixel values (0-255) at or above this are treated as
    /// "part of the subject" - see CanopySegmenter for how the mask is built.
    private let maskForegroundThreshold: UInt8 = 128

    // MARK: - Static image path (RGBA/whatever CGImage gives us; only the
    // green channel's position matters below, and it's at byte offset+1 in
    // both RGBA and BGRA layouts, so this works for standard photo decoding).

    func classify(image: UIImage) -> AerialClassification {
        guard let cgImage = image.cgImage,
              let dataProvider = cgImage.dataProvider,
              let data = dataProvider.data,
              let bytes = CFDataGetBytePtr(data) else {
            return .uncertainFallback
        }

        let width = cgImage.width
        let height = cgImage.height
        let bytesPerPixel = max(cgImage.bitsPerPixel / 8, 3)
        let bytesPerRow = cgImage.bytesPerRow
        let dataLength = CFDataGetLength(data)

        guard width > 0, height > 0, bytesPerPixel >= 3 else {
            return .uncertainFallback
        }

        var tally = PixelTally()

        for y in stride(from: 0, to: height, by: sampleStride) {
            for x in stride(from: 0, to: width, by: sampleStride) {
                let offset = y * bytesPerRow + x * bytesPerPixel
                guard offset + 2 < dataLength else { continue }

                let r = Double(bytes[offset])
                let g = Double(bytes[offset + 1])
                let b = Double(bytes[offset + 2])
                score(r: r, g: g, b: b, into: &tally)
            }
        }

        return classification(from: tally)
    }

    // MARK: - Live camera frame path (32BGRA, straight from AVCaptureVideoDataOutput)

    func classify(pixelBuffer: CVPixelBuffer, foregroundMask: CVPixelBuffer?) -> AerialClassification {
        CVPixelBufferLockBaseAddress(pixelBuffer, .readOnly)
        defer { CVPixelBufferUnlockBaseAddress(pixelBuffer, .readOnly) }
        if let foregroundMask {
            CVPixelBufferLockBaseAddress(foregroundMask, .readOnly)
        }
        defer {
            if let foregroundMask {
                CVPixelBufferUnlockBaseAddress(foregroundMask, .readOnly)
            }
        }

        guard let baseAddress = CVPixelBufferGetBaseAddress(pixelBuffer) else {
            return .uncertainFallback
        }

        let width = CVPixelBufferGetWidth(pixelBuffer)
        let height = CVPixelBufferGetHeight(pixelBuffer)
        let bytesPerRow = CVPixelBufferGetBytesPerRow(pixelBuffer)
        guard width > 0, height > 0 else {
            return .uncertainFallback
        }
        let buffer = baseAddress.assumingMemoryBound(to: UInt8.self)

        // The mask, if present, is expected to be a single-channel 8-bit
        // buffer at the SAME width/height as pixelBuffer (that's what
        // CanopySegmenter's generateScaledMaskForImage call produces). If
        // for any reason it's a different size or format than expected,
        // we defensively ignore it rather than crash or read garbage.
        var maskBase: UnsafeMutablePointer<UInt8>?
        var maskBytesPerRow = 0
        if let foregroundMask,
           CVPixelBufferGetWidth(foregroundMask) == width,
           CVPixelBufferGetHeight(foregroundMask) == height,
           let maskAddress = CVPixelBufferGetBaseAddress(foregroundMask) {
            maskBase = maskAddress.assumingMemoryBound(to: UInt8.self)
            maskBytesPerRow = CVPixelBufferGetBytesPerRow(foregroundMask)
        }

        var tally = PixelTally()

        for y in stride(from: 0, to: height, by: sampleStride) {
            for x in stride(from: 0, to: width, by: sampleStride) {
                if let maskBase {
                    let maskValue = maskBase[y * maskBytesPerRow + x]
                    guard maskValue >= maskForegroundThreshold else { continue }
                }

                let offset = y * bytesPerRow + x * 4 // BGRA = 4 bytes/pixel
                let b = Double(buffer[offset])
                let g = Double(buffer[offset + 1])
                let r = Double(buffer[offset + 2])
                score(r: r, g: g, b: b, into: &tally)
            }
        }

        return classification(from: tally)
    }

    // MARK: - Shared scoring

    private struct PixelTally {
        var scoredPixels = 0
        var colorDeadPixels = 0
        var skyBrightPixels = 0
    }

    private func score(r: Double, g: Double, b: Double, into tally: inout PixelTally) {
        let sum = r + g + b
        guard sum > canopyBrightnessFloor else { return } // skip near-black shadow

        tally.scoredPixels += 1

        let excessGreen = 2 * g - r - b
        let maxChannel = max(r, g, b)
        let minChannel = min(r, g, b)
        let saturation = maxChannel > 0 ? (maxChannel - minChannel) / maxChannel : 0

        if excessGreen < deadPixelExGThreshold && saturation < deadPixelMaxSaturation {
            tally.colorDeadPixels += 1
        }
        if sum > skyBrightnessThreshold {
            tally.skyBrightPixels += 1
        }
    }

    private func classification(from tally: PixelTally) -> AerialClassification {
        guard tally.scoredPixels > 0 else { return .uncertainFallback }

        let colorDeadFraction = Double(tally.colorDeadPixels) / Double(tally.scoredPixels)
        let porosityFraction = Double(tally.skyBrightPixels) / Double(tally.scoredPixels)
        let combined = colorWeight * colorDeadFraction + porosityWeight * porosityFraction

        let status: TreeHealthStatus
        let confidence: Double
        switch combined {
        case ..<0.35:
            status = .alive
            confidence = 1 - combined
        case 0.35..<0.6:
            status = .uncertain
            confidence = 0.5
        default:
            status = .dead
            confidence = combined
        }

        return AerialClassification(
            status: status,
            confidence: confidence,
            deadCanopyFraction: combined,
            colorDeadFraction: colorDeadFraction,
            porosityFraction: porosityFraction
        )
    }
}

private extension AerialClassification {
    static var uncertainFallback: AerialClassification {
        AerialClassification(status: .uncertain, confidence: 0, deadCanopyFraction: 0, colorDeadFraction: 0, porosityFraction: 0)
    }
}
SceneVegetationGate.swiftVegetation pre-filter 
import Vision
import CoreVideo

/// A cheap "does this frame plausibly show vegetation/outdoors at all?" check,
/// run BEFORE the per-pixel color heuristic in TreeHealthVisionAnalyzer.
///
/// Why this exists: the color heuristic alone has no concept of "tree" - it
/// just measures how green vs. brown/gray each pixel is. A couch, a wood
/// floor, or a wall satisfies "not green" exactly as well as dead canopy
/// does, so pointed indoors it will happily call your living room a dead
/// tree. This gate uses Apple's built-in Vision scene/object classifier
/// (VNClassifyImageRequest - ships on-device with iOS, no training or
/// network access required) to check whether the frame's top classifications
/// look like trees/forest/vegetation/outdoors before bothering with color
/// analysis at all.
///
/// This is still a coarse filter, not true tree detection - see README.md
/// for what a real solution (a model trained specifically on aerial canopy
/// photos) looks like.
final class SceneVegetationGate {
    private let vegetationKeywords = [
        "tree", "forest", "leaf", "leaves", "vegetation", "grass",
        "plant", "park", "woodland", "shrub", "canopy", "meadow",
        "outdoor", "field", "garden", "grove", "foliage", "farm",
        "countryside", "jungle", "orchard"
    ]
    private let minimumConfidence: Float = 0.15

    /// Checks up to the top 10 classifications Vision returns for the frame;
    /// if any plausibly vegetation/outdoor label clears the confidence bar,
    /// the frame is considered worth running the color heuristic on.
    func isLikelyVegetationScene(pixelBuffer: CVPixelBuffer) -> Bool {
        let request = VNClassifyImageRequest()
        let handler = VNImageRequestHandler(cvPixelBuffer: pixelBuffer, options: [:])

        do {
            try handler.perform([request])
        } catch {
            // Classification failed for some reason (e.g. transient Vision
            // error) - fail open rather than silently never detecting
            // anything. The color heuristic still applies its own thresholds.
            return true
        }

        guard let observations = request.results else { return true }

        for observation in observations.prefix(10) {
            guard observation.confidence >= minimumConfidence else { continue }
            let identifier = observation.identifier.lowercased()
            if vegetationKeywords.contains(where: { identifier.contains($0) }) {
                return true
            }
        }
        return false
    }
}
CanopySegmenter.swiftSubject isolation 
import Vision
import CoreVideo

/// Isolates the salient foreground subject in a frame (the tree/canopy)
/// from its background (sky, ground, neighboring objects) before the color
/// heuristic scores it, using Vision's built-in subject-lifting request
/// (`VNGenerateForegroundInstanceMaskRequest` - on-device, ships with iOS,
/// no training or network access required, same family of API behind the
/// "lift subject from photo" feature in Photos).
///
/// Why this matters: without this, TreeHealthVisionAnalyzer scores every
/// sampled pixel in the whole frame, including sky, ground, and whatever
/// else happens to be in view. That dilutes and skews the color/porosity
/// signal. With a mask, only pixels belonging to the thing Vision judged
/// to be the standalone subject of the frame get scored.
///
/// CAVEAT: I couldn't test this against real camera hardware while writing
/// it, so the mask's exact pixel format is implemented per Apple's
/// documented behavior (a single-channel 8-bit buffer, 0 = background,
/// 255 = foreground, scaled to match the source image) rather than
/// something I verified on-device. TreeHealthVisionAnalyzer defensively
/// ignores the mask if its dimensions don't match, so a format surprise
/// degrades to "score the whole frame" rather than crashing - but if you
/// notice detections behaving oddly, this is the first place to check with
/// a debug print of a few raw mask byte values.
final class CanopySegmenter {
    /// Returns a single-channel 8-bit mask the same size as `pixelBuffer`,
    /// or nil if Vision didn't find any standalone salient subject in the
    /// frame (e.g. camera pointed at an undifferentiated field of grass or
    /// open sky with nothing distinct to isolate).
    func segmentForegroundMask(pixelBuffer: CVPixelBuffer) -> CVPixelBuffer? {
        let request = VNGenerateForegroundInstanceMaskRequest()
        let handler = VNImageRequestHandler(cvPixelBuffer: pixelBuffer, options: [:])

        do {
            try handler.perform([request])
        } catch {
            return nil // fail open to "no distinct subject" rather than crash
        }

        guard let result = request.results?.first, !result.allInstances.isEmpty else {
            return nil
        }

        do {
            return try result.generateScaledMaskForImage(forInstances: result.allInstances, from: handler)
        } catch {
            return nil
        }
    }
}
CameraLiveScanner.swiftLive scan pipeline 
import AVFoundation
import UIKit
import CoreImage
import CoreLocation

/// Runs the phone's camera continuously (handheld or drone-mounted) and
/// classifies frames on the fly, looking for dead canopy. When it sees a
/// sustained "dead" signal, it fires `onAutoDetectDeadTree` so the view can
/// log a TreeRecord automatically - no manual photo capture needed.
///
/// Each sampled frame goes through two Vision-based gates before the color
/// heuristic ever runs:
///   1. `SceneVegetationGate` - is this frame even plausibly outdoors/
///      vegetation? (rules out indoor rooms, pavement, etc.)
///   2. `CanopySegmenter` - can Vision isolate a distinct standalone subject
///      (the tree) from the background? Only pixels inside that subject get
///      scored, so sky/ground/neighboring objects don't dilute the result.
/// Only after both gates pass does `HeuristicCanopyClassifier` score the
/// masked pixels on color, saturation, and canopy porosity (see
/// TreeHealthVisionAnalyzer.swift for what those mean).
///
/// Running two Vision requests per sampled frame is noticeably heavier than
/// the color math alone - if scanning feels sluggish on your device, the
/// first thing to try is raising `minFrameProcessingInterval` below.
///
/// Debounce strategy (tune the constants below to taste):
///   - Frames are only classified every `minFrameProcessingInterval` seconds,
///     not on every single frame, to keep this cheap enough to run continuously.
///   - A detection only fires after `requiredConsecutiveDeadFrames` classified
///     frames in a row come back "dead" - this filters out single-frame noise
///     (e.g. a shadow, a glare, a half-second of motion blur).
///   - After firing, it won't fire again for `minSecondsBetweenAutoLogs`
///     seconds UNLESS the phone has also moved at least
///     `minMetersBetweenAutoLogs` from where the last log happened - this
///     stops the same tree from being logged repeatedly while you hover or
///     circle over it, while still letting a fast flyover log two different
///     trees close together in time.
final class CameraLiveScanner: NSObject, ObservableObject {
    @Published var latestClassification: AerialClassification?
    @Published var isRunning = false
    @Published var consecutiveDeadFrames = 0
    /// False when the current frame doesn't look like it contains vegetation/
    /// outdoors at all (see SceneVegetationGate) - lets the UI show "no
    /// canopy in view" instead of running (and possibly misfiring) the color
    /// heuristic on indoor scenes, pavement, sky, etc.
    @Published var vegetationPresent = true
    /// False when the vegetation gate passed but Vision couldn't isolate a
    /// distinct standalone subject to score (see CanopySegmenter) - e.g. an
    /// undifferentiated field of grass or open sky with nothing to isolate.
    @Published var hasDistinctSubject = true

    let session = AVCaptureSession()

    /// Called on the main thread when a sustained dead-tree detection fires.
    /// Passes the classification and a snapshot image of the triggering frame.
    var onAutoDetectDeadTree: ((AerialClassification, UIImage?) -> Void)?
    /// Lets the scanner ask the view for the phone's current GPS coordinate
    /// (used for the distance-based part of the cooldown logic).
    var currentCoordinateProvider: (() -> CLLocationCoordinate2D?)?

    private let videoOutput = AVCaptureVideoDataOutput()
    private let processingQueue = DispatchQueue(label: "com.deadtreelogger.frame-processing")
    private let classifier: TreeHealthClassifying = HeuristicCanopyClassifier()
    private let vegetationGate = SceneVegetationGate()
    private let segmenter = CanopySegmenter()
    private let ciContext = CIContext()
    private var configured = false

    // Tuning knobs - see class doc comment above.
    // Bumped up from 0.6s now that each processed frame runs two Vision
    // requests (scene classification + subject segmentation) plus the
    // color heuristic - lower this if your device keeps up fine and you
    // want snappier detection; raise it if scanning feels sluggish.
    private let minFrameProcessingInterval: TimeInterval = 1.0
    private let requiredConsecutiveDeadFrames = 3
    private let minSecondsBetweenAutoLogs: TimeInterval = 8
    private let minMetersBetweenAutoLogs: CLLocationDistance = 5

    private var lastProcessedAt: Date = .distantPast
    private var lastAutoLogDate: Date?
    private var lastLoggedCoordinate: CLLocationCoordinate2D?

    func configureAndStart() {
        processingQueue.async { [weak self] in
            self?.configureSessionIfNeeded()
            guard let self, !self.session.isRunning else { return }
            self.session.startRunning()
            DispatchQueue.main.async { self.isRunning = true }
        }
    }

    func stop() {
        processingQueue.async { [weak self] in
            guard let self, self.session.isRunning else { return }
            self.session.stopRunning()
            DispatchQueue.main.async {
                self.isRunning = false
                self.consecutiveDeadFrames = 0
            }
        }
    }

    private func configureSessionIfNeeded() {
        guard !configured else { return }
        configured = true

        session.beginConfiguration()
        session.sessionPreset = .hd1280x720

        if let device = AVCaptureDevice.default(.builtInWideAngleCamera, for: .video, position: .back),
           let input = try? AVCaptureDeviceInput(device: device),
           session.canAddInput(input) {
            session.addInput(input)
        }

        videoOutput.videoSettings = [
            kCVPixelBufferPixelFormatTypeKey as String: kCVPixelFormatType_32BGRA
        ]
        videoOutput.alwaysDiscardsLateVideoFrames = true
        videoOutput.setSampleBufferDelegate(self, queue: processingQueue)
        if session.canAddOutput(videoOutput) {
            session.addOutput(videoOutput)
        }

        if let connection = videoOutput.connection(with: .video),
           connection.isVideoRotationAngleSupported(90) {
            connection.videoRotationAngle = 90 // portrait; this prototype assumes a fixed mount orientation
        }

        session.commitConfiguration()
    }
}

extension CameraLiveScanner: AVCaptureVideoDataOutputSampleBufferDelegate {
    func captureOutput(_ output: AVCaptureOutput, didOutput sampleBuffer: CMSampleBuffer, from connection: AVCaptureConnection) {
        let now = Date()
        guard now.timeIntervalSince(lastProcessedAt) >= minFrameProcessingInterval else { return }
        lastProcessedAt = now

        guard let pixelBuffer = CMSampleBufferGetImageBuffer(sampleBuffer) else { return }

        // Gate 1: don't even run the color heuristic on frames that don't
        // plausibly show vegetation/outdoors (indoor rooms, pavement, sky,
        // etc.) - this is what was causing "dead trees" in the living room,
        // since the color heuristic alone has no idea what a tree is.
        guard vegetationGate.isLikelyVegetationScene(pixelBuffer: pixelBuffer) else {
            DispatchQueue.main.async { [weak self] in
                self?.vegetationPresent = false
                self?.hasDistinctSubject = true
                self?.latestClassification = nil
                self?.consecutiveDeadFrames = 0
            }
            return
        }

        // Gate 2: try to isolate the standalone subject (the tree/canopy)
        // from its background before scoring it. If Vision can't find a
        // distinct subject, skip scoring this frame rather than analyzing
        // the whole scene (sky, ground, and all) undifferentiated.
        guard let mask = segmenter.segmentForegroundMask(pixelBuffer: pixelBuffer) else {
            DispatchQueue.main.async { [weak self] in
                self?.vegetationPresent = true
                self?.hasDistinctSubject = false
                self?.latestClassification = nil
                self?.consecutiveDeadFrames = 0
            }
            return
        }

        let result = classifier.classify(pixelBuffer: pixelBuffer, foregroundMask: mask)
        // Convert to a UIImage snapshot now, on this queue, while the pixel
        // buffer is still guaranteed valid - CVPixelBuffers from AVCapture
        // must not be retained/used past this callback.
        let snapshot = makeUIImage(from: pixelBuffer)

        DispatchQueue.main.async { [weak self] in
            self?.vegetationPresent = true
            self?.hasDistinctSubject = true
            self?.handle(result: result, snapshot: snapshot)
        }
    }

    private func handle(result: AerialClassification, snapshot: UIImage?) {
        latestClassification = result
        consecutiveDeadFrames = (result.status == .dead) ? consecutiveDeadFrames + 1 : 0

        guard consecutiveDeadFrames >= requiredConsecutiveDeadFrames else { return }

        let currentCoordinate = currentCoordinateProvider?()
        if let lastDate = lastAutoLogDate, Date().timeIntervalSince(lastDate) < minSecondsBetweenAutoLogs {
            let movedFarEnough: Bool
            if let last = lastLoggedCoordinate, let current = currentCoordinate {
                let distance = CLLocation(latitude: last.latitude, longitude: last.longitude)
                    .distance(from: CLLocation(latitude: current.latitude, longitude: current.longitude))
                movedFarEnough = distance >= minMetersBetweenAutoLogs
            } else {
                movedFarEnough = false // no location fix - fall back to time-only cooldown
            }
            guard movedFarEnough else { return }
        }

        lastAutoLogDate = Date()
        lastLoggedCoordinate = currentCoordinate
        consecutiveDeadFrames = 0

        onAutoDetectDeadTree?(result, snapshot)
    }

    private func makeUIImage(from pixelBuffer: CVPixelBuffer) -> UIImage? {
        let ciImage = CIImage(cvPixelBuffer: pixelBuffer)
        guard let cgImage = ciContext.createCGImage(ciImage, from: ciImage.extent) else { return nil }
        return UIImage(cgImage: cgImage)
    }
}

import UIKit

/// Saves/loads aerial photos as JPEGs in the app's Documents directory, since
/// SwiftData records store the filename, not the image bytes.
enum PhotoStore {
    static func save(image: UIImage) -> String? {
        guard let data = image.jpegData(compressionQuality: 0.8) else { return nil }
        let fileName = "\(UUID().uuidString).jpg"
        let url = FileManager.default
            .urls(for: .documentDirectory, in: .userDomainMask)[0]
            .appendingPathComponent(fileName)
        do {
            try data.write(to: url)
            return fileName
        } catch {
            print("PhotoStore: failed to save image - \(error)")
            return nil
        }
    }

    static func load(fileName: String) -> UIImage? {
        let url = FileManager.default
            .urls(for: .documentDirectory, in: .userDomainMask)[0]
            .appendingPathComponent(fileName)
        guard let data = try? Data(contentsOf: url) else { return nil }
        return UIImage(data: data)
    }
}

/// Continuous aerial scanning: point the camera (handheld, or mounted on a
/// drone) at tree canopy and this view classifies frames in the background,
/// automatically logging a TreeRecord whenever it sees a sustained dead-tree
/// signal. There's no manual "take a photo" step - see CameraLiveScanner for
/// the detection/debounce logic that decides when to auto-log.
struct AerialScanView: View {
    @Environment(\.modelContext) private var modelContext
    @StateObject private var location = LocationManager()
    @StateObject private var scanner = CameraLiveScanner()
    @State private var cameraAuthorized = false
    @State private var loggedThisSessionCount = 0
    @State private var lastLogMessage: String?

    var body: some View {
        NavigationStack {
            ZStack {
                if cameraAuthorized {
                    CameraPreviewView(session: scanner.session)
                        .ignoresSafeArea()
                } else {
                    Color.black.ignoresSafeArea()
                    ContentUnavailableView(
                        "Camera access needed",
                        systemImage: "camera",
                        description: Text("Enable camera access in Settings to use continuous aerial scanning.")
                    )
                    .foregroundStyle(.white)
                }

                VStack {
                    statusBanner
                    Spacer()
                    controls
                }
                .padding()
            }
            .navigationTitle("Aerial Scan")
            .navigationBarTitleDisplayMode(.inline)
            .onAppear {
                location.requestPermission()
                requestCameraAccessIfNeeded()
                scanner.currentCoordinateProvider = { location.currentLocation }
                scanner.onAutoDetectDeadTree = { classification, snapshot in
                    autoLog(classification: classification, snapshot: snapshot)
                }
            }
            .onDisappear { scanner.stop() }
        }
    }

    private var statusBanner: some View {
        VStack(alignment: .leading, spacing: 4) {
            HStack {
                Circle()
                    .fill(bannerColor)
                    .frame(width: 12, height: 12)
                Text(bannerText)
                    .font(.subheadline.bold())
                    .lineLimit(1)
                Spacer()
                Text("Logged: \(loggedThisSessionCount)")
                    .font(.caption)
            }
            // Live readout of the current frame's signals, useful for tuning
            // the thresholds in TreeHealthVisionAnalyzer.swift against real
            // healthy/dead trees - point at a known-healthy tree and a
            // known-dead one and see what numbers come back.
            if scanner.vegetationPresent, scanner.hasDistinctSubject, let result = scanner.latestClassification {
                Text("Combined: \(Int(result.deadCanopyFraction * 100))%  ·  color: \(Int(result.colorDeadFraction * 100))%  ·  porosity: \(Int(result.porosityFraction * 100))%")
                    .font(.caption2)
                    .foregroundStyle(.white.opacity(0.8))
            }
        }
        .padding(10)
        .background(.black.opacity(0.55), in: RoundedRectangle(cornerRadius: 12))
        .foregroundStyle(.white)
    }

    private var bannerColor: Color {
        guard scanner.isRunning else { return .gray }
        guard scanner.vegetationPresent, scanner.hasDistinctSubject else { return .gray }
        switch scanner.latestClassification?.status {
        case .dead: return .red
        case .alive: return .green
        default: return .orange
        }
    }

    private var bannerText: String {
        guard scanner.isRunning else { return "Scanning paused" }
        guard scanner.vegetationPresent else { return "No canopy in view" }
        guard scanner.hasDistinctSubject else { return "No distinct subject to score" }
        guard let result = scanner.latestClassification else { return "Analyzing..." }
        switch result.status {
        case .dead: return "Dead canopy in frame (\(scanner.consecutiveDeadFrames)/3 to confirm)"
        case .alive: return "Healthy canopy"
        case .uncertain: return "Uncertain"
        }
    }

    private var controls: some View {
        VStack(spacing: 8) {
            if let message = lastLogMessage {
                Text(message)
                    .font(.caption)
                    .padding(8)
                    .background(.black.opacity(0.55), in: Capsule())
                    .foregroundStyle(.white)
            }
            Button(scanner.isRunning ? "Stop Scanning" : "Start Scanning") {
                if scanner.isRunning {
                    scanner.stop()
                } else {
                    scanner.configureAndStart()
                }
            }
            .buttonStyle(.borderedProminent)
            .tint(scanner.isRunning ? .red : .green)
            .disabled(!cameraAuthorized)
        }
    }

    private func requestCameraAccessIfNeeded() {
        switch AVCaptureDevice.authorizationStatus(for: .video) {
        case .authorized:
            cameraAuthorized = true
        case .notDetermined:
            AVCaptureDevice.requestAccess(for: .video) { granted in
                DispatchQueue.main.async { cameraAuthorized = granted }
            }
        default:
            cameraAuthorized = false
        }
    }

    private func autoLog(classification: AerialClassification, snapshot: UIImage?) {
        let coordinate = location.currentLocation
        let record = TreeRecord(
            timestamp: .now,
            latitude: coordinate?.latitude ?? 0,
            longitude: coordinate?.longitude ?? 0,
            source: .aerial,
            status: classification.status,
            confidence: classification.confidence,
            notes: "Auto-logged during live scan, dead canopy fraction \(Int(classification.deadCanopyFraction * 100))%"
        )
        if let snapshot, let fileName = PhotoStore.save(image: snapshot) {
            record.photoFileName = fileName
        }
        modelContext.insert(record)
        loggedThisSessionCount += 1

        if let coordinate {
            lastLogMessage = String(format: "Logged dead tree at %.5f, %.5f", coordinate.latitude, coordinate.longitude)
        } else {
            lastLogMessage = "Logged dead tree (no GPS fix yet)"
        }
    }
}

void takeReadingAndNotify() {
  digitalWrite(STATUS_LED_PIN, HIGH);


  for (int i = 0; i < 8; i++) {

  }

  unsigned long startTime = micros();
  unsigned long elapsed = 0;
  int peakAmplitude = 0;
  bool detected = false;

  while (elapsed < TIMEOUT_US) {
    int reading = analogRead(PIEZO_RX_PIN);
    if (reading > peakAmplitude) peakAmplitude = reading;
    if (reading >= RX_TRIGGER_LEVEL) {
      detected = true;
      break;
    }
    elapsed = micros() - startTime;
  }

  float velocity;
  int status; // 0 alive, 1 dead, 2 uncertain

  if (!detected) {
   
    velocity = 0;
    status = 2;
  } else {
    float seconds = elapsed / 1000000.0;
    float meters = TRUNK_DISTANCE_CM / 100.0;
    velocity = meters / seconds;

    if (velocity < DEAD_VELOCITY_THRESHOLD_MS - UNCERTAIN_BAND_MS) {
      status = 1; // dead
    } else if (velocity > DEAD_VELOCITY_THRESHOLD_MS + UNCERTAIN_BAND_MS) {
      status = 0; // alive
    } else {
      status = 2; // uncertain, too close to the threshold to call confidently
    }
  }

  char payload[96];
  snprintf(payload, sizeof(payload), "{\"v\":%.1f,\"amp\":%d,\"status\":%d}",
           velocity, peakAmplitude, status);

  readingCharacteristic->setValue(payload);
  readingCharacteristic->notify();

  Serial.println(payload);
  digitalWrite(STATUS_LED_PIN, LOW);
}
